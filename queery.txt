/*
MongoDB basics

Start MongoDB service (macOS/Homebrew):
    brew services start mongodb-community

Open Mongo shell:
    mongosh

Use database:
    use mydb

Stop MongoDB service:
    brew services stop mongodb-community

Note:
    - Tables are called collections in MongoDB.
    - Rows/columns are represented as documents/fields.

Basic commands
    show dbs
    use mydb
    db.test.insertOne({ name: "Salah" })
    db.test.find()

Aggregation: match + addFields + project + out
    db.test.aggregate([
        {
            $match: {
                gender: "male",
                age: { $gt: 25 }
            }
        },
        {
            $addFields: {
                course: "AI Agent",
                grade: 3.8
            }
        },
        {
            $project: {
                name: 1,
                course: 1,
                grade: 1
            }
        },
        {
            $out: "student-course"
            // Alternative: $merge: "student-course"
        }
    ])

Aggregation: group by friend and keep selected fields
    db.test.aggregate([
        {
            $group: {
                _id: "$friends",
                count: { $sum: 1 },
                fullDoc: { $push: "$$ROOT" }
            }
        },
        {
            $project: {
                count: 1,
                "fullDoc.name": 1,
                "fullDoc.email": 1,
                "fullDoc.phone": 1
            }
        }
    ])

Aggregation: total salary
    db.test.aggregate([
        {
            $group: {
                _id: null,
                totalSalary: { $sum: "$salary" }
            }
        }
    ])

Aggregation: unwind interests then group by age
    db.test.aggregate([
        { $unwind: "$interests" },
        {
            $group: {
                _id: "$age",
                interestsPerAge: { $push: "$interests" }
            }
        }
    ])

Aggregation: bucket + sort + limit + project
    db.test.aggregate([
        {
            $bucket: {
                groupBy: "$age",
                boundaries: [0, 20, 26, 30],
                default: "After 30",
                output: {
                    count: { $sum: 1 },
                    fullInfo: { $push: "$$ROOT" }
                }
            }
        },
        { $sort: { count: -1 } },
        { $limit: 2 },
        { $project: { count: 1, _id: 1 } }
    ])

Aggregation: $facet (multiple pipelines at once)
    db.test.aggregate([
        {
            $facet: {
                skillsDeko: [
                    { $unwind: "$skills" },
                    { $group: { _id: "$skills", count: { $sum: 1 } } }
                ],
                friendsDeko: [
                    { $unwind: "$friends" },
                    { $group: { _id: "$friends", count: { $sum: 1 } } }
                ],
                interestsDeko: [
                    { $unwind: "$interests" },
                    { $group: { _id: "$interests", count: { $sum: 1 } } }
                ]
            }
        },
        {
            $project: {
                skills: "$skillsDeko",
                friends: "$friendsDeko",
                interests: "$interestsDeko",
                _id: 0
            }
        }
    ])

$lookup (referencing)
    db.orders.aggregate([
        {
            $lookup: {
                from: "test",
                localField: "userId",
                foreignField: "_id",
                as: "userInfo"
            }
        },
        {
            $project: {
                _id: 1,
                userId: 1,
                products: 1,
                userInfo: 1
            }
        }
    ])

Indexing examples
    // Create index
    db.getCollection("test").createIndex({ email: 1 })

    // Drop index
    db.getCollection("test").dropIndex({ email: 1 })

    // Create text index
    db.getCollection("test").createIndex({ friends: "text" })

    // Text search
    db.getCollection("test")
        .find({ $text: { $search: "Alice" } })
        .project({ friends: 1 })
